<script>
const items=document.querySelectorAll(".reagent-item");
const nextBtn=document.getElementById("nextBtn");
const autoBtn=document.getElementById("autoBtn");
const resetBtn=document.getElementById("resetBtn");
const tempSlider=document.getElementById("tempSlider");
const tempText=document.getElementById("tempText");
const stageText=document.getElementById("stageText");
const cycleEl=document.getElementById("cycle");
const dnaNumEl=document.getElementById("dnaNum");
const tooltip=document.getElementById("tooltip");
const primer1=document.getElementById("primer1");
const primer2=document.getElementById("primer2");
const required=["template","primer","taq","dntp","buffer"];
const added=new Set();
let cycle=0,copies=1,autoRun=false,stageIdx=0;
const stages=[
{name:"变性",targetT:95,desc:"95℃ 变性：双链DNA解旋分开"},
{name:"退火",targetT:55,desc:"50–55℃ 退火：引物结合互补序列"},
{name:"延伸",targetT:72,desc:"72℃ 延伸：Taq聚合酶合成新链"}
];
items.forEach(el=>{
el.onclick=()=>{
const id=el.dataset.id;
if(added.has(id)){
added.delete(id);
el.classList.remove("added");
}else{
added.add(id);
el.classList.add("added");
}
checkReady();
}
})
function checkReady(){
const ok=required.every(k=>added.has(k));
nextBtn.disabled=!ok;
autoBtn.disabled=!ok;
}
tempSlider.oninput=()=>{
tempText.textContent=`${tempSlider.value} ℃`;
}
document.querySelectorAll(".mol").forEach(mol=>{
mol.onmouseenter=(e)=>{
tooltip.textContent=mol.dataset.info;
tooltip.style.display="block";
}
mol.onmousemove=(e)=>{
tooltip.style.left=(e.clientX+12)+"px";
tooltip.style.top=(e.clientY+12)+"px";
}
mol.onmouseleave=()=>{
tooltip.style.display="none";
}
})
nextBtn.onclick=()=>{
runStep();
}
async function runStep(){
const s=stages[stageIdx];
stageText.textContent=s.desc;
primer1.setAttribute("opacity",1);
primer2.setAttribute("opacity",1);
primer1.setAttribute("d","M 170 145 L 210 145");
primer2.setAttribute("d","M 130 175 L 170 175");
await sleep(1000);
stageIdx++;
if(stageIdx>=stages.length){
cycle++;
copies*=2;
cycleEl.textContent=cycle;
dnaNumEl.textContent=copies;
stageIdx=0;
stageText.textContent="本轮扩增完成，可以继续下一轮";
}
}
function sleep(ms){
return new Promise(res=>setTimeout(res,ms));
}
autoBtn.onclick=async ()=>{
if(autoRun) return;
autoRun=true;
while(autoRun && !nextBtn.disabled){
await runStep();
await sleep(1200);
}
autoRun=false;
}
resetBtn.onclick=()=>{
added.clear();
items.forEach(i=>i.classList.remove("added"));
cycle=0;
copies=1;
stageIdx=0;
autoRun=false;
cycleEl.textContent=0;
dnaNumEl.textContent=1;
tempSlider.value=25;
tempText.textContent="25 ℃（室温）";
stageText.textContent="先配齐所有试剂，再开始 PCR！";
primer1.setAttribute("opacity",0);
primer2.setAttribute("opacity",0);
checkReady();
}
</script>
